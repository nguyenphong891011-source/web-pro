volatile long pulseLeft = 0, pulseRight = 0; void IRAM_ATTR isrL() { pulseLeft++; } void IRAM_ATTR isrR() { pulseRight++; } // ==================== CẢM BIẾN 12 MẮT (CD74HC4067) ==================== // LƯU Ý: Trên schematic chân SIG là IO32, nếu mạch thực tế nối IO34 thì giữ 34, // nếu theo đúng schematic gốc thì đổi SIG_PIN thành 32. #define NS 12 const uint8_t MUX_S[4] = {19, 18, 17, 16}; // Hệ số D (chống văng khi bẻ góc) int MAX_SPD = 210; // Tốc độ tối đa trên đoạn thẳng (0..255) int MIN_SPD = 105; // Tốc độ nền thấp nhất khi vào cua gắt float K_DECEL = 0.022f; // Hệ số tự động giảm tốc khi |error| tăng int TURN_OUT = 175; // Tốc độ bánh trong khi mất line ở góc nhọn int DASH_ERR = 1800; // Ngưỡng |error| tối đa để nhận diện là đang đi thẳng vào Nét Đứt int DASH_MAX_MS = 450; // Thời gian tối đa phóng thẳng qua khoảng trắng nét đứt (ms) int BRAKE_MS = 20; // Thời gian phanh ngược khử quán tính khi lao từ thẳng vào cua gắt (ms) int STOP_CNT2 = 1; // Số lần gặp vạch đen toàn bộ (sau thời gian IGNORE_MS) để dừng int IGNORE_MS = 1800; // Bỏ qua vạch đen ngang đầu tiên sau START trong X (ms) int SENSOR_THR = 450; // Ngưỡng nhận diện mắt thấy line (0..1000) int speed_dk = 130; // Tốc độ chạy Manual // ==================== BIẾN TRẠNG THÁI CHẠY ==================== int lastError = 0; float filteredD = 0; int currentSpd = 0; IGNORE_MS = prefs.getInt("igms", IGNORE_MS); SENSOR_THR = prefs.getInt("sthr", SENSOR_THR); prefs.end(); } void saveParams() { prefs.begin("robot", false); prefs.putFloat("kp", KP); prefs.putFloat("kd", KD); prefs.putInt("maxs", MAX_SPD); prefs.putInt("mins", MIN_SPD); prefs.putFloat("kdec", K_DECEL); prefs.putInt("tout", TURN_OUT); prefs.putInt("tin", TURN_IN); prefs.putInt("sout", SPIN_OUT); prefs.putInt("sin", SPIN_IN); } // ==================== ĐỌC CẢM BIẾN & CALIB ==================== void readRaw() { for (int i = 0; i < NS; i++) { for (int b = 0; b < 4; b++) digitalWrite(MUX_S[b], (MUX_CH[i] >> b) & 1); delayMicroseconds(120); // Tối ưu tốc độ quét mux analogRead(SIG_PIN); rawv[i] = (analogRead(SIG_PIN) + analogRead(SIG_PIN)) >> 1; } } void resetCal() { for (int i = 0; i < NS; i++) { calMin[i] = 4095; calMax[i] = 0; } } void calibrateStep() { readRaw(); for (int i = 0; i < NS; i++) { if (rawv[i] < calMin[i]) calMin[i] = rawv[i]; if (rawv[i] > calMax[i]) calMax[i] = rawv[i]; } } bool channelOk(int i) { return calMax[i] > calMin[i] + 250; } int docSensor() { readRaw(); uint32_t s = 0, w = 0; int cnt = 0; pattern = 0; for (int i = 0; i < NS; i++) { int range = (int)calMax[i] - (int)calMin[i]; int v = 0; if (range > 50) { v = constrain(((long)rawv[i] - calMin[i]) * 1000 / range, 0, 1000); if (INVERT_COLOR) v = 1000 - v; } if (v > SENSOR_THR) { pattern |= (1 << i); cnt++; } if (v > 100) { s += (uint32_t)v * POS[i]; w += v; } } if (w > 0) lastLinePos = s / w; return cnt; } // ==================== ĐIỀU KHIỂN ĐỘNG CƠ & PHANH TB6612 ==================== void pwmSetup() { #if ESP_ARDUINO_VERSION_MAJOR >= 3 ledcAttach(R_PWM, PWM_FREQ, PWM_RES); R_PWM : L_PWM, duty); #else ledcWrite(right ? 0 : 1, duty); #endif } void setMotor(bool right, int sp) { sp = constrain(sp, -255, 255); bool fwd = (sp >= 0) ^ (right ? R_INVERT : L_INVERT); int in1 = right ? R_IN1 : L_IN1; int in2 = right ? R_IN2 : L_IN2; if (sp == 0) { // Phanh điện từ (Short Brake) trên TB6612FNG: IN1 = HIGH, IN2 = HIGH digitalWrite(in1, HIGH); digitalWrite(in2, HIGH); pwmWrite(right, 255); } else { digitalWrite(in1, fwd ? HIGH : LOW); digitalWrite(in2, fwd ? LOW : HIGH); pwmWrite(right, abs(sp)); } } void move(int left, int right) { setMotor(false, left); setMotor(true, right); } void stopMotor() { move(0, 0); } // Phanh xung ngược khử quán tính khi từ đoạn thẳng tốc độ cao lao vào góc cua void activeBrakePulse() { if (BRAKE_MS > 0 && millis() - highSpeedStraightMs < 150) { move(-160, -160); delay(BRAKE_MS); highSpeedStraightMs = 0; } } // ==================== BLE GIAO TIẾP ==================== void bleSend(const char* msg) { if (deviceConnected && pTx) { pTx->setValue((uint8_t*)msg, strlen(msg)); pTx->notify(); delay(3); } Serial.println(msg); } void sendAllParams() { char b[240]; snprintf(b, sizeof(b), "CFG:%.4f,%.4f,%d,%d,%.4f,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d", KP, KD, MAX_SPD, MIN_SPD, K_DECEL, TURN_OUT, TURN_IN, SPIN_OUT, SPIN_IN, DASH_ERR, DASH_MAX_MS, BRAKE_MS, STOP_CNT2, IGNORE_MS, SENSOR_THR); bleSend(b); } void sendStatus() { if (!deviceConnected || millis() - lastStatusTime < STATUS_INTERVAL) return; lastStatusTime = millis(); char pat[NS + 1]; for (int i = 0; i < NS; i++) pat[i] = (pattern & (1 << i)) ? '1' : '0'; pat[NS] = 0; unsigned long elapsed = runLine ? (millis() - runStartMs) : finishMs; char buf[240]; snprintf(buf, sizeof(buf), "{\"mode\":\"%s\",\"case\":\"%s\",\"pos\":%u,\"pat\":\"%s\",\"spd\":%d,\"err\":%d,\"cal\":%d,\"c1\":%d,\"c2\":%d,\"ms\":%lu}", currentModeName.c_str(), currentCaseName.c_str(), lastLinePos, pat, currentSpd, lastError, calibrating ? 2 : (calibrated ? 1 : 0), cnt1, cnt2, elapsed); pTx->setValue((uint8_t*)buf, strlen(buf)); pTx->notify(); } // ==================== GIẢI THUẬT DÒ LINE TỐC ĐỘ CAO ==================== void finishRun() { move(-180, -180); delay(25); // Phanh ngược nhẹ để đứng im ngay trong ô Đích stopMotor(); runLine = false; finishMs = millis() - runStartMs; currentSpd = 0; currentModeName = "DONE"; currentCaseName = "FINISH"; char msg[64]; snprintf(msg, sizeof(msg), "LOG:VE DICH! %.2fs", finishMs / 1000.0f); bleSend(msg); } void lineFollowPID() { int cnt = docSensor(); // Kiểm tra nhóm mắt rìa trái (mắt 0, 1) và rìa phải (mắt 10, 11) bool edgeL = (pattern & 0b000000000011) != 0; bool edgeR = (pattern & 0b110000000000) != 0; // 1. KIỂM TRA VẠCH ĐEN NGANG / Ô ĐÍCH (Nhiều mắt cùng đen: >= 10 mắt) if (cnt >= 10) { if (!blackLatched) { blackLatched = true; // Bỏ qua vạch chữ thập đầu tiên ngay sau vạch START if (millis() - runStartMs > (unsigned long)IGNORE_MS) { cnt2++; if (cnt2 >= STOP_CNT2) { finishRun(); return; } } } // Nếu đang băng qua vạch chữ thập đầu tiên thì giữ chạy thẳng currentCaseName = "CROSS_LINE"; currentSpd = MAX_SPD; move(MAX_SPD, MAX_SPD); return; } else { blackLatched = false; } // 2. XỬ LÝ KHI MẤT LINE (000000000000): PHÂN BIỆT NÉT ĐỨT vs VĂNG CUA GẮT if (cnt == 0) { if (!whiteLatched) { whiteLatched = true; whiteStartMs = millis(); // Nếu trước khi mất line, xe đang ở gần tâm và không chạm mắt rìa -> NÉT ĐỨT! if (abs(lastError) <= DASH_ERR && !rememberEdgeL && !rememberEdgeR) { inDashMode = true; cnt1++; } else { inDashMode = false; } } // Nhánh 2A: Đang bay qua đoạn NÉT ĐỨT -> Giữ tốc độ cao & bám thẳng tâm! if (inDashMode && (millis() - whiteStartMs < (unsigned long)DASH_MAX_MS)) { currentCaseName = "DASH_LINE"; currentSpd = MAX_SPD; int trim = (int)(lastError * KP * 0.25f); // Giữ hướng nhẹ theo sai số cuối move(constrain(MAX_SPD + trim, -255, 255), constrain(MAX_SPD - trim, -255, 255)); return; } // Nhánh 2B: Mất line do văng khỏi góc cua 90 độ / đỉnh răng cưa -> Xoay bắt line gắt! currentCaseName = "SPIN_RECOVERY"; currentSpd = SPIN_OUT; if (rememberEdgeR || lastError > 0) { move(SPIN_OUT, SPIN_IN); // Xoay phải gắt } else { move(SPIN_IN, SPIN_OUT); // Xoay trái gắt } return; } // Đã thấy lại line -> Reset trạng thái mất line whiteLatched = false; inDashMode = false; if (edgeL && !edgeR) { rememberEdgeL = true; rememberEdgeR = false; } else if (edgeR && !edgeL) { rememberEdgeR = true; rememberEdgeL = false; } else if (cnt <= 4 && abs((int)lastLinePos - SETPOINT) < 1500) { rememberEdgeL = false; rememberEdgeR = false; } // 3. BÁM LINE PID KẾT HỢP TỐC ĐỘ ĐỘNG (DYNAMIC SPEED: THẲNG NHANH - CUA CHẬM) int error = (int)lastLinePos - SETPOINT; int dErr = error - lastError; filteredD = 0.6f * filteredD + 0.4f * dErr; // Lọc nhiễu khâu D lastError = error; // Tính tốc độ nền tự động giảm khi |error| hoặc |dErr| lớn (vào cua) int dynamicBase = MAX_SPD - (int)(abs(error) * K_DECEL) - (int)(abs(filteredD) * 0.04f); dynamicBase = constrain(dynamicBase, MIN_SPD, MAX_SPD); currentSpd = dynamicBase; if (abs(error) < 1000) { currentCaseName = "STRAIGHT_MAX"; highSpeedStraightMs = millis(); } else { currentCaseName = "CORNER_PID"; } int adjust = (int)(error * KP + filteredD * KD); int L = constrain(dynamicBase + adjust, -255, 255); int R = constrain(dynamicBase - adjust, -255, 255); move(L, R); } // ==================== XỬ LÝ LỆNH TỪ WEB BLE ==================== void handleBLECommand(String cmd) { cmd.trim(); if (cmd.startsWith("SPD:")) { speed_dk = constrain(cmd.substring(4).toInt(), 30, 255); char b[32]; snprintf(b, sizeof(b), "LOG:SPD=%d", speed_dk); bleSend(b); return; } if (cmd == "TEST") { testMode = !testMode; runLine = false; stopMotor(); bleSend(testMode ? "LOG:TEST_ON" : "LOG:TEST_OFF"); return; } if (cmd == "CAL") { runLine = false; testMode = false; stopMotor(); calibrated = false; calibrating = true; calEnd = millis() + CAL_TIME; resetCal(); pattern = 0; return; } if (cmd == "GETPID" || cmd == "GETCFG") { sendAllParams(); return; } if (cmd == "AUTO") { if (!calibrated) { bleSend("LOG:CHUA CALIB CAM BIEN!"); return; } testMode = false; runLine = true; lastError = 0; filteredD = 0; lastLinePos = SETPOINT; pulseLeft = pulseRight = 0; cnt1 = 0; cnt2 = 0; whiteLatched = false; blackLatched = false; inDashMode = false; rememberEdgeL = false; rememberEdgeR = false;
